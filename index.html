/* ============================================================ MIS Portal & Workspace — app.js CHỈ xử lý hành vi hiển thị/ẩn thuần giao diện: thu gọn sidebar, menu mobile, chuyển theme sáng/tối, đóng/mở modal, và khởi tạo biểu đồ cho panel Performance & Resource Metrics. KHÔNG chỉnh sửa/thêm/xoá bất kỳ dữ liệu nào trong bảng hay KPI. ============================================================ */ document.addEventListener('DOMContentLoaded', () => { /* ---------- 1. Sidebar collapse (desktop) ---------- */ const sidebar = document.getElementById('app-sidebar'); const collapseBtn = document.getElementById('sidebar-collapse-btn'); const applyCollapsedState = (collapsed) => { sidebar.classList.toggle('collapsed', collapsed); const icon = collapseBtn.querySelector('i'); icon.className = collapsed ? 'ri-menu-unfold-line' : 'ri-menu-fold-line'; }; applyCollapsedState(localStorage.getItem('mis-sidebar-collapsed') === 'true'); collapseBtn?.addEventListener('click', () => { const collapsed = !sidebar.classList.contains('collapsed'); applyCollapsedState(collapsed); localStorage.setItem('mis-sidebar-collapsed', String(collapsed)); }); /* ---------- 2. Sidebar mobile drawer ---------- */ const mobileToggleBtn = document.getElementById('nav-toggle-mobile-btn'); let backdrop = document.querySelector('.sidebar-backdrop'); if (!backdrop) { backdrop = document.createElement('div'); backdrop.className = 'sidebar-backdrop'; document.body.appendChild(backdrop); } const closeMobileNav = () => { sidebar.classList.remove('mobile-open'); backdrop.classList.remove('active'); }; mobileToggleBtn?.addEventListener('click', () => { sidebar.classList.add('mobile-open'); backdrop.classList.add('active'); }); backdrop.addEventListener('click', closeMobileNav); /* ---------- 3. Theme toggle (sáng / tối) ---------- */ const themeToggleBtn = document.getElementById('theme-toggle-btn'); const htmlEl = document.documentElement; const applyTheme = (theme) => { htmlEl.setAttribute('data-theme', theme); const icon = themeToggleBtn.querySelector('i'); icon.className = theme === 'dark' ? 'ri-sun-line' : 'ri-moon-line'; }; applyTheme(localStorage.getItem('mis-theme') || 'light'); themeToggleBtn?.addEventListener('click', () => { const next = htmlEl.getAttribute('data-theme') === 'dark' ? 'light' : 'dark'; applyTheme(next); localStorage.setItem('mis-theme', next); }); /* ---------- 4. Điều hướng sidebar (chỉ đổi trạng thái active hiển thị) ---------- */ const navLinks = document.querySelectorAll('.nav-item-link'); navLinks.forEach((link) => { link.addEventListener('click', (event) => { event.preventDefault(); navLinks.forEach((l) => l.closest('.nav-item')?.classList.remove('active')); link.closest('.nav-item')?.classList.add('active'); closeMobileNav(); }); }); /* ---------- 5. Modal: Add New Record ---------- */ const modalOverlay = document.getElementById('record-modal-overlay'); const addRecordBtn = document.getElementById('add-record-btn'); const cancelModalBtn = document.getElementById('cancel-modal-btn'); const closeModalBtn = document.getElementById('modal-close-btn'); const recordForm = document.getElementById('record-form'); const openModal = () => { modalOverlay.classList.add('active'); document.getElementById('form-department')?.focus(); }; const closeModal = () => { modalOverlay.classList.remove('active'); recordForm?.reset(); }; addRecordBtn?.addEventListener('click', openModal); cancelModalBtn?.addEventListener('click', closeModal); closeModalBtn?.addEventListener('click', closeModal); modalOverlay?.addEventListener('click', (event) => { if (event.target === modalOverlay) closeModal(); }); document.addEventListener('keydown', (event) => { if (event.key === 'Escape' && modalOverlay.classList.contains('active')) { closeModal(); } }); /* ---------- 6. Submit form: chỉ đóng modal (không chỉnh sửa dữ liệu bảng) ---------- */ recordForm?.addEventListener('submit', (event) => { event.preventDefault(); closeModal(); }); /* ---------- 7. Chart: Performance & Resource Metrics ---------- */ const chartCanvas = document.getElementById('analytics-chart'); if (chartCanvas && window.Chart) { const ctx = chartCanvas.getContext('2d'); const gradientA = ctx.createLinearGradient(0, 0, 0, 300); gradientA.addColorStop(0, 'rgba(99, 102, 241, 0.28)'); gradientA.addColorStop(1, 'rgba(99, 102, 241, 0)'); const gradientB = ctx.createLinearGradient(0, 0, 0, 300); gradientB.addColorStop(0, 'rgba(6, 182, 212, 0.22)'); gradientB.addColorStop(1, 'rgba(6, 182, 212, 0)'); new Chart(ctx, { type: 'line', data: { labels: ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'], datasets: [ { label: 'System Load', data: [62, 58, 71, 66, 78, 54, 60], borderColor: '#6366f1', backgroundColor: gradientA, borderWidth: 2.5, tension: 0.4, fill: true, pointRadius: 0, pointHoverRadius: 5, pointHoverBackgroundColor: '#6366f1', }, { label: 'Throughput', data: [40, 45, 42, 55, 48, 60, 52], borderColor: '#06b6d4', backgroundColor: gradientB, borderWidth: 2.5, tension: 0.4, fill: true, pointRadius: 0, pointHoverRadius: 5, pointHoverBackgroundColor: '#06b6d4', }, ], }, options: { responsive: true, maintainAspectRatio: false, interaction: { mode: 'index', intersect: false }, plugins: { legend: { position: 'top', align: 'end', labels: { usePointStyle: true, pointStyle: 'circle', boxWidth: 7, font: { family: 'Inter', size: 12 }, color: '#64748b', }, }, tooltip: { backgroundColor: '#0f172a', padding: 10, cornerRadius: 8, titleFont: { family: 'Inter', weight: '600' }, bodyFont: { family: 'Inter' }, }, }, scales: { x: { grid: { display: false }, ticks: { color: '#94a3b8', font: { family: 'Inter', size: 11 } }, }, y: { grid: { color: 'rgba(148, 163, 184, 0.15)' }, ticks: { color: '#94a3b8', font: { family: 'Inter', size: 11 } }, }, }, }, }); } });
